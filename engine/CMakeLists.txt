option(SEEDX_FETCH_LLAMA_CPP "Fetch pinned llama.cpp source when local inference is enabled" ON)
if(SEEDX_ENABLE_LLAMA_CPP)
 if(SEEDX_FETCH_LLAMA_CPP)
  include(FetchContent)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b4514 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(llama_cpp)
 else()
  find_package(llama CONFIG REQUIRED)
 endif()
 set(SEEDX_INFERENCE_SOURCE src/llama_backend.cpp)
else()
 set(SEEDX_INFERENCE_SOURCE src/llama_backend_stub.cpp)
endif()
add_library(seed_engine STATIC src/agent_kernel.cpp src/guardian.cpp src/policy.cpp src/audit_log.cpp src/event_bus.cpp src/local_ai.cpp src/online_ai.cpp ${SEEDX_INFERENCE_SOURCE})
target_include_directories(seed_engine PUBLIC include)
target_compile_features(seed_engine PUBLIC cxx_std_20)
target_compile_options(seed_engine PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
if(SEEDX_ENABLE_LLAMA_CPP)
 target_link_libraries(seed_engine PUBLIC llama)
 target_compile_definitions(seed_engine PUBLIC SEEDX_HAS_LLAMA_CPP=1)
endif()
