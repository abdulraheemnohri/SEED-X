cmake_minimum_required(VERSION 3.22.1)
project(seedx_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID)
 message(FATAL_ERROR "Configure this target with the Android NDK.")
endif()
set(GLUE_DIR "${ANDROID_NDK}/sources/android/native_app_glue")
if(NOT EXISTS "${GLUE_DIR}/android_native_app_glue.c")
 message(FATAL_ERROR "android_native_app_glue.c missing from configured NDK.")
endif()
add_library(android_native_app_glue STATIC "${GLUE_DIR}/android_native_app_glue.c")
target_include_directories(android_native_app_glue PUBLIC "${GLUE_DIR}")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../../engine" "${CMAKE_CURRENT_BINARY_DIR}/seed_engine")
add_library(seedx_android SHARED native_main.cpp native_ui.cpp)
target_include_directories(seedx_android PRIVATE "${GLUE_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(seedx_android PRIVATE seed_engine android_native_app_glue android log EGL GLESv3)
target_compile_options(seedx_android PRIVATE -Wall -Wextra -Wpedantic)
